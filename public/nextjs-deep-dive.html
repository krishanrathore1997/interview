Caching ArchitecturePerformance
4-layer cache model, unstable_cache, revalidation strategies
🗂
The 4 Cache Layers
1. Request Memoization (per-request)
// Same fetch/cache() call deduped per request
// Automatic — no config needed
// Cleared after each request completes

const getPost = cache(async (id) => {
  return db.find(id)  // 1 query even if called 10x
})
2. Data Cache (persistent, cross-request)
// Stored on server, survives deploys (on Vercel)
// Only fetch() calls are cached here by default
// Invalidated by revalidateTag / revalidatePath

await fetch(url, { next: { tags: ['posts'], revalidate: 3600 } })
3. Full Route Cache (build-time, static)
// Pre-rendered HTML + RSC payload
// Static routes cached at build
// Bypassed when: cookies(), headers(),
//   searchParams, dynamic = 'force-dynamic'

export const revalidate = 86400 // ISR: 24h
4. Router Cache (client-side, session)
// In-memory cache of RSC payloads
// Prefetched Link targets stored here
// Cleared on: full page reload, router.refresh()
// Duration: static 5min, dynamic 30sec

router.refresh()  // force clear client cache

export const getCachedPosts = unstable_cache(

    revalidate: 3600,

// Cache scope tags
import { cacheTag, cacheLife } from 'next/cache'
async function getUser(id) {
  'use cache'
  cacheTag(`user-${id}`)   // granular tag
  cacheLife('hours')         // 'seconds'|'minutes'|'hours'|'days'|'weeks'|'max'
  return db.user.find(id)
}

Revalidation Strategies
// Time-based ISR
export const revalidate = 60  // in page.tsx

// On-demand — from Server Action or API route
import { revalidatePath, revalidateTag } from 'next/cache'

// Purge all cached data for /blog path
revalidatePath('/blog')
revalidatePath('/blog/[slug]', 'page')    // specific dynamic pages
revalidatePath('/dashboard', 'layout')   // layout + all children
revalidatePath('/', 'layout')            // ALL pages (nuclear)

// Purge by tag (more targeted)
revalidateTag('posts')         // all data tagged 'posts'
revalidateTag(`post-${id}`)   // specific post

// Webhook-triggered revalidation (API route)
// POST /api/revalidate?tag=posts&secret=TOKEN
export async function POST(req: Request) {
  const { searchParams } = new URL(req.url)
  const tag = searchParams.get('tag')
  if (tag) revalidateTag(tag)
  return Response.json({ revalidated: true })
}

import { revalidatePath } from 'next/cache'
import { redirect } from 'next/navigation'
import { z } from 'zod'

const schema = z.object({
  title: z.string().min(1).max(255),
  body:  z.string().min(10),
})

export async function createPost(prevState: any, formData: FormData) {
  const result = schema.safeParse({
    title: formData.get('title'),
    body:  formData.get('body'),
  })

  if (!result.success) {
    return { errors: result.error.flatten().fieldErrors }
  }

  const user = await getAuthUser()     // always verify auth in actions
  await prisma.post.create({ data: { ...result.data, userId: user.id } })
  revalidatePath('/posts')
  redirect('/posts')
}

import Credentials from 'next-auth/providers/credentials'
import { PrismaAdapter } from '@auth/prisma-adapter'

export const { handlers, auth, signIn, signOut } = NextAuth({
  adapter: PrismaAdapter(prisma),
  providers: [
    GitHub({ clientId: process.env.GITHUB_ID!, clientSecret: process.env.GITHUB_SECRET! }),
    Credentials({
      async authorize(credentials) {
        const user = await prisma.user.findUnique({ where: { email: credentials.email as string } })
        if (!user) return null
        const valid = await bcrypt.compare(credentials.password as string, user.password)
        return valid ? user : null
      }
    }),
  ],
  callbacks: {
    async session({ session, token }) {
      session.user.id   = token.sub!
      session.user.role = token.role as string
      return session
    },
    async jwt({ token, user }) {
      if (user) token.role = (user as any).role
      return token
    },
  },
  session: { strategy: 'jwt' },
})

// middleware.ts (root)
import { auth } from './auth'
import { NextResponse } from 'next/server'

export default auth(function middleware(req) {
  const isAuth  = !!req.auth
  const isAdmin = req.auth?.user?.role === 'admin'
  const path   = req.nextUrl.pathname

  // Protect /dashboard routes
  if (path.startsWith('/dashboard') && !isAuth)
    return NextResponse.redirect(new URL('/login', req.url))

  // Admin-only routes
  if (path.startsWith('/admin') && !isAdmin)
    return NextResponse.redirect(new URL('/403', req.url))
})

// store/useCartStore.ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'

interface CartStore {
  items: CartItem[]
  addItem: (item: CartItem) => void
  removeItem: (id: string) => void
  total: () => number
}

export const useCartStore = create<CartStore>()(
  persist(
    (set, get) => ({
      items: [],
      addItem: (item) => set((s) => ({ items: [...s.items, item] })),
      removeItem: (id) => set((s) => ({ items: s.items.filter(i => i.id !== id) })),
      total: () => get().items.reduce((s, i) => s + i.price, 0),
    }),
    { name: 'cart-storage' }  // persists to localStorage
  )
)

// Usage (Client Component only)

// Reusable API auth helper
export async function withAuth(req: NextRequest, handler: Function) {
  const token = req.headers.get('authorization')?.replace('Bearer ', '')
  if (!token) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  const user = await verifyToken(token)
  return handler(req, user)
}
