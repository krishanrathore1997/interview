class EventEmitter {
  // Private fields (ES2022) — truly private
  #listeners = new Map()
  #count = 0

  // Static field + method
  static instances = 0
  static create() { return new this() }

  // Private method
  #validate(event) { return typeof event === 'string' }

  on(event, cb) {
    if (!this.#validate(event)) throw new TypeError('Event must be a string')
    if (!this.#listeners.has(event)) this.#listeners.set(event, [])
    this.#listeners.get(event).push(cb)
    return this  // chainable
  }

  emit(event, ...args) {
    this.#listeners.get(event)?.forEach(cb => cb(...args))
    return this
  }
}

function fetchWithTimeout(url, ms = 5000) {
  const controller = new AbortController()
  const id = setTimeout(() => controller.abort(), ms)
  return fetch(url, { signal: controller.signal })
    .finally(() => clearTimeout(id))

// Retry with exponential backoff
async function withRetry(fn, retries = 3, delay = 500) {
  for (let i = 0; i <= retries; i++) {
    try { return await fn() }
    catch (err) {
      if (i === retries) throw err
      await new Promise(r => setTimeout(r, delay * 2 ** i))
    }
  }
}

Concurrency Control
// Rate-limit concurrent requests
async function pLimit(tasks, concurrency) {
  const results = []
  const executing = new Set()
  for (const task of tasks) {
    const p = task().finally(() => executing.delete(p))
    results.push(p)
    executing.add(p)
    if (executing.size >= concurrency)
      await Promise.race(executing)
  }
  return Promise.all(results)
}
// Usage: await pLimit(urls.map(u => () => fetch(u)), 3)

// Queue class for sequential processing
class AsyncQueue {
  #queue = []
  #running = false
  add(task) {
    return new Promise((resolve, reject) => {
      this.#queue.push({ task, resolve, reject })
      this.#run()
    })
  }
  async #run() {
    if (this.#running) return
    this.#running = true
    while (this.#queue.length) {
      const { task, resolve, reject } = this.#queue.shift()
      await task().then(resolve).catch(reject)
    }
    this.#running = false
  }
}

// Memoization
function memoize(fn) {
  const cache = new Map()
  return function(...args) {
    const key = JSON.stringify(args)
    if (cache.has(key)) return cache.get(key)
    const result = fn.apply(this, args)
    cache.set(key, result)
    return result
  }
}

// Curry — auto-curried
const curry = (fn) =>
  (function curried(...args) {
    return args.length >= fn.length
      ? fn(args)
      : (...more) => curried(...args, ...more)
  })

// Observer (pub/sub)
class EventBus {
  #events = new Map()
  on(e, fn)  { (this.#events.get(e) ?? this.#events.set(e, new Set()).get(e)).add(fn) }
  off(e, fn) { this.#events.get(e)?.delete(fn) }

// Builder pattern
class QueryBuilder {
  #table = ''; #conditions = []; #limit = null
  from(t)    { this.#table = t; return this }
  where(c)   { this.#conditions.push(c); return this }
  limit(n)   { this.#limit = n; return this }
  build()    {
    let q = `SELECT * FROM ${this.#table}`
    if (this.#conditions.length) q += ` WHERE ${this.#conditions.join(' AND ')}`
    if (this.#limit) q += ` LIMIT ${this.#limit}`
    return q
  }
}

// useLocalStorage
function useLocalStorage<T>(key: string, initial: T) {
  const [value, setValue] = useState<T>(() => {
    try { return JSON.parse(localStorage.getItem(key) ?? '') }
    catch { return initial }
  })
  const set = useCallback((v: T) => {
    setValue(v)
    localStorage.setItem(key, JSON.stringify(v))
  }, [key])
  return [value, set] as const
}

// useDebounce
function useDebounce<T>(value: T, delay: number): T {
  const [debounced, setDebounced] = useState(value)
  useEffect(() => {
    const id = setTimeout(() => setDebounced(value), delay)
    return () => clearTimeout(id)
  }, [value, delay])
  return debounced
}

// useIntersectionObserver
function useInView(ref: RefObject<Element>, options?: IntersectionObserverInit) {
  const [inView, setInView] = useState(false)
  useEffect(() => {
    const obs = new IntersectionObserver(([e]) => setInView(e.isIntersecting), options)
    if (ref.current) obs.observe(ref.current)
    return () => obs.disconnect()
  }, [])
  return inView
}

function useWorker(workerUrl: string) {
  const workerRef = useRef<Worker>()
  useEffect(() => {
    workerRef.current = new Worker(new URL(workerUrl, import.meta.url), { type: 'module' })
    return () => workerRef.current?.terminate()
  }, [])
  return useCallback((data) => {
    return new Promise((res) => {
      workerRef.current!.postMessage(data)
      workerRef.current!.onmessage = ({ data }) => res(data)
    })
  }, [])
}

const TabCtx = React.createContext<{ active: string; setActive: (s: string) => void }>(null!)

function Tabs({ children, defaultTab }) {
  const [active, setActive] = useState(defaultTab)
  return <TabCtx.Provider value={{ active, setActive }}>{children}</TabCtx.Provider>
}

Tabs.Tab = function Tab({ id, children }) {
  const { active, setActive } = use(TabCtx)
  return <button onClick={() => setActive(id)} data-active={active === id}>{children}</button>
}

Tabs.Panel = function Panel({ id, children }) {
  const { active } = use(TabCtx)
  return active === id ? <div>{children}</div> : null
}

class ErrorBoundary extends React.Component {
  state = { error: null }
  static getDerivedStateFromError(err) { return { error: err } }
  componentDidCatch(err, info) { logError(err, info.componentStack) }
  render() {
    return this.state.error
      ? <ErrorFallback error={this.state.error} reset={() => this.setState({ error: null })} />
      : this.props.children
  }
}

Headless UI Pattern
💡Headless = all behaviour/state, no UI. Consumer provides the rendering. Maximum flexibility.
// Headless useCombobox hook
function useCombobox<T>({ items, getLabel }: { items: T[], getLabel: (i: T) => string }) {
  const [query,    setQuery]    = useState('')
  const [selected, setSelected] = useState<T | null>(null)
  const [isOpen,   setIsOpen]   = useState(false)

  const filtered = useMemo(() =>
    items.filter(i => getLabel(i).toLowerCase().includes(query.toLowerCase())),
    [items, query]
  )

  return {
    // Prop getters — spread onto elements
    getInputProps: () => ({
      value: query,
      onChange: (e) => { setQuery(e.target.value); setIsOpen(true) },
      onFocus: () => setIsOpen(true),
    }),
    getItemProps: (item: T) => ({
      onClick: () => { setSelected(item); setQuery(getLabel(item)); setIsOpen(false) },
    }),
    filtered, selected, isOpen,
  }
}

// Usage — consumer owns ALL styling
const { getInputProps, getItemProps, filtered, isOpen } = useCombobox({ items, getLabel })
<input {...getInputProps()} className="my-input" />
