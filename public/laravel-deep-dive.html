Attribute API (Laravel 9+)
use Illuminate\Database\Eloquent\Casts\Attribute;

class User extends Model {
    // Combined get+set accessor
    protected function firstName(): Attribute {
        return Attribute::make(
            get: fn($v) => ucfirst($v),
            set: fn($v) => strtolower($v),
        );
    }

    // Computed from multiple columns
    protected function fullName(): Attribute {
        return Attribute::make(
            get: fn() => "{$this->first_name} {$this->last_name}",
        )->shouldCache(); // cache computed value
    }

    // Custom cast class
    protected function casts(): array {
        return [
            'password'        => 'hashed',        // auto-hash
            'email_verified'  => 'datetime',
            'settings'        => 'array',
            'balance'         => 'decimal:2',
            'is_admin'        => 'boolean',
            'role'            => UserRole::class, // PHP Enum
            'address'         => AddressCast::class,
        ];
    }
}

Chunking, Cursors & Lazy Collections
⚠Never use all() on large tables — it loads everything into memory at once.
// chunk — processes in batches of N, memory safe
Post::chunk(200, function (Collection $posts) {
    foreach ($posts as $post) {
        processPost($post);
    }
});

// chunkById — safest when modifying records during iteration
Post::chunkById(100, function ($posts) {
    $posts->each->update(['processed' => true]);
});

// cursor — PHP generator, one model in memory at a time
foreach (Post::cursor() as $post) {
    processPost($post);
}

// lazy — lazy collection, best of both worlds
Post::lazy()->each(function ($post) {
    processPost($post);
});

// lazyById — cursor + chunks, safest for large datasets
Post::lazyById()->filter(fn($p) => $p->needsUpdate())->each(...);

N+1 Detection & Prevention
💥N+1 is the #1 Eloquent performance killer. 100 posts × 1 user query = 101 queries.
// ENABLE strict mode in AppServiceProvider (throws on N+1)
Model::preventLazyLoading(! app()->isProduction());

// BAD — triggers N+1 (one query per post)
$posts = Post::all();
foreach ($posts as $post) {
    echo $post->user->name; // ← LAZY load = N queries
}

// GOOD — eager load relationships
$posts = Post::with('user')->get(); // 2 queries total

// Nested eager loading
Post::with('comments.author', 'tags', 'user.profile')->get();

// withCount — adds *_count column, no extra model hydration
Post::withCount(['comments', 'likes'])->get();
// $post->comments_count, $post->likes_count

// withSum / withAvg / withMin / withMax
Post::withSum('likes', 'count')->get();
// $post->likes_sum_count

Job Chaining & Batching
Chain — sequential execution
// Run jobs in order — next only starts if previous succeeds
Bus::chain([
    new ProcessPayment($order),
    new SendReceipt($order),
    new UpdateInventory($order),
])->catch(function (Throwable $e) {
    // fires if ANY job fails
})->dispatch();
Batch — parallel execution with progress
$batch = Bus::batch([
    new ImportRow($row1),
    new ImportRow($row2),
    new ImportRow($row3),
])->then(function (Batch $batch) {
    // all jobs completed
})->catch(function (Batch $batch, Throwable $e) {
    // first failure
})->finally(function (Batch $batch) {
    // always runs when batch finishes
})->name('CSV Import')
   ->onQueue('imports')
   ->allowFailures()   // don't cancel on single failure
   ->dispatch();

// Check batch progress
$batch = Bus::findBatch($batch->id);
$batch->progress();   // 0–100
$batch->finished();
$batch->failedJobs;

Retries, Timeouts & Failure Handling
class ProcessPayment implements ShouldQueue
{
    use InteractsWithQueue, Queueable;

    public $tries           = 5;      // max attempts
    public $timeout         = 120;   // seconds before force-kill
    public $maxExceptions   = 3;     // distinct exceptions before fail
    public $backoff         = [1, 5, 10]; // seconds between retries

    // Exponential backoff
    public function backoff(): array
    { return [2, 4, 8, 16, 32]; }

    public function retryUntil(): DateTime
    { return now()->addHours(4); } // retry for up to 4h

    public function failed(Throwable $exception): void
    {
        // Called after all retries exhausted
        Log::critical('Payment failed', ['error' => $exception->getMessage()]);
        $this->order->update(['status' => 'failed']);
    }

    public function handle(): void
    {
        if ($this->order->isPaid()) {
            return $this->delete(); // idempotency check
        }
        // ... process payment
    }
}

class GenerateReport implements ShouldQueue, ShouldBeUnique

// (released when job starts, not finishes)
class SyncUserData implements ShouldQueue, ShouldBeUniqueUntilProcessing {}

// WithoutOverlapping middleware — lock while running
public function middleware(): array
{
    return [
        (new WithoutOverlapping($this->user->id))
            ->dontRelease()         // discard if overlapping
            ->expireAfter(300),    // lock expires in 5 min
        new RateLimited('emails'), // rate limit the job itself
    ];
}

class OrderStatusUpdated implements ShouldBroadcast
{
    use Dispatchable, SerializesModels;

    public function __construct(public Order $order) {}

    // Which channel(s) to broadcast on
    public function broadcastOn(): array
    {
        return [
            new PrivateChannel('orders.'.$this->order->id),
            new Channel('public-orders'),        // public
            new PresenceChannel('order-room'),  // presence
        ];
    }

    // Customize broadcast payload
    public function broadcastWith(): array
    { return ['id' => $this->order->id, 'status' => $this->order->status]; }

    // Custom event name on frontend
    public function broadcastAs(): string
    { return 'order.status.updated'; }

    // Only broadcast if condition true
    public function broadcastWhen(): bool
    { return $this->order->status !== 'draft'; }
}

Action Pattern
💡Actions are single-responsibility classes — one action does one thing. Replaces bloated controllers and services.
// app/Actions/Posts/CreatePost.php
class CreatePost
{
    public function __construct(
        private readonly PostRepository $posts,
        private readonly TagService     $tags,
    ) {}

    public function handle(User $user, CreatePostData $data): Post
    {
        $post = $this->posts->create([
            'title'   => $data->title,
            'body'    => $data->body,
            'user_id' => $user->id,
            'slug'    => Str::slug($data->title),
        ]);

        $this->tags->syncTagsForPost($post, $data->tags);
        PostCreated::dispatch($post);
        return $post;
    }
}

// In Controller (thin — no business logic)
public function store(StorePostRequest $req, CreatePost $action): JsonResponse
{
    $post = $action->handle($req->user(), CreatePostData::from($req->validated()));
    return new PostResource($post);
}

DTOs & Data Classes
💡DTOs (Data Transfer Objects) make data flow explicit and type-safe — no more $request->all() magic arrays.
// PHP 8.x readonly properties (perfect for DTOs)
readonly class CreatePostData
{
    public function __construct(
        public string  $title,
        public string  $body,
        public bool    $published     = false,
        public array   $tags          = [],
        public ?string $publishAt     = null,
    ) {}

    public static function from(array $data): static
    {
        return new static(
            title:     $data['title'],
            body:      $data['body'],
            published: $data['published'] ?? false,
            tags:      $data['tags']       ?? [],
        );
    }
}

// In Job — serializes properly due to readonly
class CreatePostJob implements ShouldQueue {
    public function __construct(public readonly CreatePostData $data) {}

// Header versioning (Accept header)
class ApiVersionMiddleware
{
    public function handle(Request $request, Closure $next): Response
    {
        $version = $request->header('Accept-Version', 'v1');
        app()->instance('api.version', $version);
        return $next($request);
    }
}

// Versioned Resources
class PostResource extends JsonResource {
    public function toArray(Request $req): array {
        return match (app('api.version')) {
            'v2'    => $this->toV2Array(),
            default => $this->toV1Array(),
        };
    }
}

Conditional Requests & HATEOAS
// Conditional fields — resource only includes when loaded
public function toArray(Request $request): array
{
    return [
        'id'       => $this->id,
        'title'    => $this->title,

        // Only when relationship is loaded
        'author'   => UserResource::make($this->whenLoaded('user')),

        // Only when pivot data is loaded
        'order'    => $this->whenPivotLoaded('post_tag', fn() => $this->pivot->order),

        // Only for authenticated users
        'my_vote'  => $this->when($request->user(), fn() =>
            $this->votes->where('user_id', $request->user()->id)->first()),

        // Only for admins
        'ip_addr'  => $this->when($request->user()?->isAdmin(), $this->ip_address),

        // Merge array conditionally
        $this->mergeWhen($request->has('include_stats'), [
            'views' => $this->views,
            'likes' => $this->likes_count,
        ]),

        // HATEOAS links
        'links' => [
            'self'     => route('api.posts.show',    $this->id),
            'comments' => route('api.posts.comments', $this->id),
        ],
    ];
}

// ApiResponse trait for controllers
trait ApiResponses
{
    protected function success($data, int $status = 200): JsonResponse {
        return response()->json([
            'data'    => $data,
            'success' => true,
        ], $status);
    }

    protected function error(string $message, int $status): JsonResponse {
        return response()->json([
            'error'   => $message,
            'success' => false,
            'status'  => $status,
        ], $status);
    }
