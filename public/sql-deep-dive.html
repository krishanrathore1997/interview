SELECT — Query Anatomy
💡SQL evaluates in this order: FROM → JOIN → WHERE → GROUP BY → HAVING → SELECT → DISTINCT → ORDER BY → LIMIT
-- Full query structure (logical execution order in comments)
SELECT                      -- 6. what to return
    u.id,
    u.name,
    COUNT(o.id)    AS order_count,
    SUM(o.total)  AS lifetime_value
FROM   users    u          -- 1. base table
JOIN   orders   o  ON u.id = o.user_id  -- 2. join
WHERE  u.active = true     -- 3. filter rows
GROUP BY u.id, u.name     -- 4. group
HAVING COUNT(o.id) > 2    -- 5. filter groups
ORDER BY lifetime_value DESC  -- 7. sort
LIMIT  10                  -- 8. paginate
OFFSET 20;                 -- skip 20

-- SELECT DISTINCT
SELECT DISTINCT country FROM users;

-- SELECT INTO (create table from query)
SELECT * INTO users_backup FROM users;

WHERE (a = 1 OR b = 2) AND c = 3  -- use parens!

CASE Expression — Conditional Logic
-- Simple CASE (equality checks)
SELECT
  name,
  CASE status
    WHEN 'A' THEN 'Active'
    WHEN 'I' THEN 'Inactive'
    ELSE           'Unknown'
  END AS status_label
FROM users;

-- Searched CASE (any condition)
SELECT
  order_id,
  total,
  CASE
    WHEN total >= 1000 THEN 'Platinum'
    WHEN total >=  500 THEN 'Gold'
    WHEN total >=  100 THEN 'Silver'
    ELSE                    'Bronze'
  END AS tier
FROM orders;

-- CASE inside aggregate (conditional count)
SELECT
  COUNT(CASE WHEN status = 'paid'    THEN 1 END) AS paid_count,
  COUNT(CASE WHEN status = 'pending' THEN 1 END) AS pending_count,
  SUM(CASE WHEN status = 'paid'    THEN amount ELSE 0 END) AS paid_total
FROM orders;

-- IIF (SQL Server shorthand)
IIF(score >= 60, 'Pass', 'Fail')
-- PostgreSQL equivalent:
NULLIF(a, b)   -- returns NULL if a = b
COALESCE(a, b, c) -- first non-NULL value

JOIN Types — Visual Reference
-- INNER JOIN — only matching rows in BOTH tables
SELECT u.name, o.total
FROM  users  u
JOIN  orders o ON u.id = o.user_id;
-- users WITHOUT orders are EXCLUDED

-- LEFT JOIN — all left rows, NULLs for non-matching right
SELECT u.name, o.total
FROM  users  u
LEFT JOIN orders o ON u.id = o.user_id;
-- users WITHOUT orders appear with o.total = NULL

-- Find rows with NO match (anti-join pattern)
SELECT u.name
FROM  users  u
LEFT JOIN orders o ON u.id = o.user_id
WHERE o.id IS NULL;  -- users who never ordered

-- FULL OUTER JOIN — all rows from both tables
SELECT a.name, b.name
FROM table_a a
FULL OUTER JOIN table_b b ON a.id = b.a_id;

-- CROSS JOIN — every combination (cartesian product)
SELECT s.size, c.color
FROM sizes s CROSS JOIN colors c;
-- 4 sizes × 6 colors = 24 rows

SELF JOIN & Multiple JOINs
-- SELF JOIN — table joins itself (hierarchical data)
SELECT
  e.name  AS employee,
  m.name  AS manager
FROM      employees e
LEFT JOIN employees m ON e.manager_id = m.id;

-- Multiple JOINs — chained
SELECT
  o.id         AS order_id,
  u.name       AS customer,
  p.name       AS product,
  cat.name     AS category,
  oi.quantity
FROM      orders      o
JOIN      users       u   ON o.user_id    = u.id
JOIN      order_items oi  ON o.id         = oi.order_id
JOIN      products    p   ON oi.product_id = p.id
LEFT JOIN categories  cat ON p.category_id = cat.id
WHERE o.status = 'completed';

-- JOIN with multiple conditions
JOIN prices p
  ON p.product_id = oi.product_id
 AND p.effective_date <= o.created_at
 AND (p.end_date IS NULL OR p.end_date > o.created_at)

LATERAL JOIN & APPLYAdvanced
💡LATERAL (PG/MySQL 8) / CROSS APPLY (SQL Server) lets the right side reference the left side — like a correlated subquery that returns multiple rows.
-- Get last 3 orders per user (without window fn)
SELECT u.id, u.name, recent.order_id, recent.total
FROM users u
CROSS JOIN LATERAL (
  SELECT id AS order_id, total
  FROM   orders
  WHERE  user_id = u.id          -- references u!
  ORDER BY created_at DESC
  LIMIT 3
) recent;

-- LEFT JOIN LATERAL — include users with no orders
FROM users u
LEFT JOIN LATERAL (...) recent ON TRUE

-- LATERAL with function (PG: unnest, json_each)
SELECT t.id, tag.value
FROM  tasks t
CROSS JOIN LATERAL unnest(t.tags) AS tag(value);

-- SQL Server equivalent: CROSS APPLY / OUTER APPLY
FROM users u
CROSS APPLY (SELECT TOP 3 * FROM orders WHERE user_id = u.id ORDER BY id DESC) r

Aggregate Functions
-- Core aggregates
SELECT
  COUNT(*)           AS total_rows,
  COUNT(email)        AS non_null_emails,  -- ignores NULLs
  COUNT(DISTINCT country) AS unique_countries,
  SUM(amount)          AS total,
  AVG(amount)          AS average,
  MIN(amount)          AS minimum,
  MAX(amount)          AS maximum
FROM orders;

-- FILTER clause (SQL:2003, PG / SQLite / DuckDB)
-- Cleaner than CASE WHEN inside aggregate
SELECT
  COUNT(*)                                  AS total,
  COUNT(*) FILTER (WHERE status = 'paid')    AS paid,
  SUM(amount) FILTER (WHERE status = 'paid') AS paid_total
FROM orders;

-- STRING_AGG — aggregate strings (PG, SQL Server 2017+)
SELECT user_id,
  STRING_AGG(tag, ', ' ORDER BY tag) AS all_tags
FROM user_tags
GROUP BY user_id;
-- MySQL: GROUP_CONCAT(tag ORDER BY tag SEPARATOR ', ')

-- ARRAY_AGG (PostgreSQL)
ARRAY_AGG(product_id ORDER BY quantity DESC)

ROLLUP, CUBE & GROUPING SETS
💡These generate subtotals and grand totals in one query — no UNION ALL hacks needed.
-- ROLLUP — hierarchical subtotals
SELECT
  COALESCE(region,   'ALL REGIONS')  AS region,
  COALESCE(category, 'ALL CATS')     AS category,
  SUM(revenue)
FROM sales
GROUP BY ROLLUP(region, category);
-- Produces: region+category, region subtotal, grand total

-- CUBE — all combinations of subtotals
GROUP BY CUBE(region, category, year)

-- GROUPING SETS — explicit groups
GROUP BY GROUPING SETS (
  (region, category),  -- group 1
  (region),            -- group 2
  ()                   -- grand total
)

-- GROUPING() — detect subtotal rows
SELECT
  CASE WHEN GROUPING(region) = 1
       THEN 'Total' ELSE region END AS region,
  SUM(revenue)
FROM sales
GROUP BY ROLLUP(region);

Window Function Anatomy
💡Window functions compute across a set of rows related to the current row — without collapsing rows like GROUP BY does.
function_name(column) OVER (
  PARTITION BY partition_col   -- optional: reset for each group
  ORDER BY     order_col       -- optional: order within window
  ROWS BETWEEN start AND end   -- optional: frame
)

-- Frame options:
ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW  -- default running total
ROWS BETWEEN 2 PRECEDING AND 2 FOLLOWING          -- 5-row moving avg
ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING   -- from current to end
RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW  -- RANGE vs ROWS

-- Named WINDOW (reuse)
SELECT
  ROW_NUMBER()   OVER w,
  SUM(amount)   OVER w
FROM orders
WINDOW w AS (PARTITION BY user_id ORDER BY created_at);
🏆
Ranking Functions
SELECT
  name, department, salary,

  -- ROW_NUMBER: unique 1,2,3... (no ties)
  ROW_NUMBER()  OVER (PARTITION BY dept ORDER BY salary DESC) AS rn,

  -- RANK: gaps after ties (1,1,3)
  RANK()        OVER (PARTITION BY dept ORDER BY salary DESC) AS rnk,

  -- DENSE_RANK: no gaps (1,1,2)
  DENSE_RANK()  OVER (PARTITION BY dept ORDER BY salary DESC) AS dr,

  -- NTILE: split into N buckets (quartiles, deciles)
  NTILE(4)      OVER (ORDER BY salary)                      AS quartile,

  -- PERCENT_RANK: percentile (0.0 – 1.0)
  PERCENT_RANK() OVER (ORDER BY salary)                   AS pct_rank

FROM employees;

-- Top N per group (classic pattern)
SELECT * FROM (
  SELECT *,
    ROW_NUMBER() OVER (PARTITION BY dept ORDER BY salary DESC) AS rn
  FROM employees
) ranked
WHERE rn <= 3;  -- top 3 earners per department
📈
Analytic & Running Totals
SELECT
  order_date,
  amount,

  -- Running total
  SUM(amount) OVER (
    PARTITION BY user_id
    ORDER BY     order_date
    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
  ) AS running_total,

  -- 7-day moving average
  AVG(amount) OVER (
    ORDER BY order_date
    ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
  ) AS moving_avg_7d,

  -- Lag/Lead — access adjacent rows
  LAG(amount)  OVER (ORDER BY order_date)             AS prev_amount,
  LEAD(amount) OVER (ORDER BY order_date)             AS next_amount,
  LAG(amount, 2, 0) OVER (ORDER BY order_date)        AS two_ago,

  -- MoM growth
  ROUND(
    (amount - LAG(amount) OVER (ORDER BY order_date))
    / NULLIF(LAG(amount) OVER (ORDER BY order_date), 0) * 100, 2
  ) AS growth_pct,

  -- FIRST_VALUE / LAST_VALUE / NTH_VALUE
  FIRST_VALUE(amount) OVER (PARTITION BY user_id ORDER BY order_date) AS first_order

FROM orders;

CTE Patterns
-- Basic CTE (readability over performance)
WITH active_users AS (
  SELECT id, name, email
  FROM   users
  WHERE  active = true
    AND  created_at > NOW() - INTERVAL '90 days'
)
SELECT * FROM active_users WHERE email LIKE '%@gmail%';

-- Chained CTEs
WITH
monthly_sales AS (
  SELECT
    DATE_TRUNC('month', created_at) AS month,
    SUM(amount)                       AS revenue
  FROM orders
  WHERE status = 'paid'
  GROUP BY 1
),
with_growth AS (
  SELECT
    month, revenue,
    LAG(revenue) OVER (ORDER BY month) AS prev_revenue
  FROM monthly_sales         -- reference first CTE
)
SELECT
  month, revenue,
  ROUND((revenue - prev_revenue) / NULLIF(prev_revenue, 0) * 100, 1) AS growth_pct
FROM with_growth
ORDER BY month;

Recursive CTEs — Hierarchies & Sequences
-- Org chart — walk employee hierarchy
WITH RECURSIVE org_chart AS (
  -- Anchor: start with CEO (no manager)
  SELECT id, name, manager_id, 0 AS depth,
         name::text AS path
  FROM  employees
  WHERE manager_id IS NULL

  UNION ALL

  -- Recursive: join children to parents
  SELECT e.id, e.name, e.manager_id,
         oc.depth + 1,
         oc.path || ' > ' || e.name
  FROM  employees  e
  JOIN  org_chart  oc ON e.manager_id = oc.id
)
SELECT
  REPEAT('  ', depth) || name AS org_chart,
  depth, path
FROM org_chart
ORDER BY path;

-- Generate date series with recursive CTE
WITH RECURSIVE dates (d) AS (
  SELECT '2024-01-01'::date
  UNION ALL
  SELECT d + 1 FROM dates WHERE d < '2024-12-31'
)
SELECT * FROM dates;
-- PG shortcut: generate_series('2024-01-01','2024-12-31','1 day'::interval)

Subquery Types
-- Scalar subquery — returns single value
SELECT
  name,
  salary,
  (SELECT AVG(salary) FROM employees) AS avg_salary,
  salary - (SELECT AVG(salary) FROM employees) AS diff
FROM employees;

-- Correlated subquery — references outer query row-by-row
SELECT name, department, salary
FROM  employees e
WHERE salary > (
  SELECT AVG(salary)
  FROM   employees
  WHERE  department = e.department   -- references outer e!
);

-- Derived table (inline view)
SELECT dept_stats.dept, dept_stats.avg_sal
FROM (
  SELECT department AS dept, AVG(salary) AS avg_sal
  FROM   employees
  GROUP BY department
) dept_stats
WHERE dept_stats.avg_sal > 70000;

-- ANY / ALL
WHERE salary > ANY (SELECT salary FROM managers)  -- > at least one
WHERE salary > ALL (SELECT salary FROM managers)  -- > every one

EXISTS vs IN — Performance
⚠NOT IN fails silently if the subquery contains NULLs. Always prefer NOT EXISTS for anti-joins.
-- EXISTS: short-circuits on first match (fast for large sets)
SELECT * FROM customers c
WHERE EXISTS (
  SELECT 1            -- SELECT 1 is convention, SELECT * also works
  FROM   orders o
  WHERE  o.customer_id = c.id
);

-- NOT EXISTS: safe anti-join (handles NULLs correctly)
SELECT * FROM customers c
WHERE NOT EXISTS (
  SELECT 1 FROM orders o
  WHERE o.customer_id = c.id
);

-- IN: ok for small, static lists
WHERE country_code IN ('US', 'GB', 'IN')

-- NOT IN: DANGEROUS if subquery can return NULL
-- NOT IN (1, 2, NULL) = always FALSE!
-- Use NOT EXISTS or LEFT JOIN ... IS NULL instead

-- UNION / UNION ALL / INTERSECT / EXCEPT
SELECT id FROM table_a
UNION     -- removes duplicates (like DISTINCT)
SELECT id FROM table_b;

UNION ALL    -- keeps duplicates, faster
INTERSECT    -- rows in BOTH
EXCEPT       -- rows in first but NOT second (MINUS in Oracle)

CREATE TABLE — Full Syntax
CREATE TABLE orders (
  -- Primary key
  id           BIGSERIAL    PRIMARY KEY,  -- PG auto-increment
  -- id    BIGINT IDENTITY(1,1) PRIMARY KEY -- SQL Server
  -- id    INT AUTO_INCREMENT PRIMARY KEY    -- MySQL

  -- UUID primary key (better for distributed systems)
  -- id UUID DEFAULT gen_random_uuid() PRIMARY KEY,

  -- Foreign key
  user_id      BIGINT       NOT NULL
                             REFERENCES users(id)
                             ON DELETE CASCADE
                             ON UPDATE RESTRICT,

  -- Strings
  status       VARCHAR(20) NOT NULL DEFAULT 'pending',
  notes        TEXT,

  -- Numbers
  amount       NUMERIC(12,2) NOT NULL CHECK (amount >= 0),
  quantity     INT          NOT NULL DEFAULT 1,

  -- Dates
  created_at   TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  updated_at   TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  deleted_at   TIMESTAMPTZ,              -- nullable = soft delete

  -- JSON (PG)
  metadata     JSONB        DEFAULT '{}',

  -- Table-level constraint
  CONSTRAINT orders_status_check
    CHECK (status IN ('pending', 'paid', 'cancelled')),
  UNIQUE (user_id, reference_no)  -- composite unique
);

Views, Materialized Views & Triggers
-- View — virtual table (query stored)
CREATE OR REPLACE VIEW active_user_orders AS
  SELECT u.id, u.name, COUNT(o.id) AS order_count
  FROM users u
  JOIN orders o ON u.id = o.user_id
  WHERE u.active = true
  GROUP BY u.id, u.name;

-- Materialized View (PG) — stores results physically
CREATE MATERIALIZED VIEW monthly_stats AS
  SELECT DATE_TRUNC('month', created_at) AS month,
         COUNT(*), SUM(amount)
  FROM  orders GROUP BY 1
WITH DATA;
REFRESH MATERIALIZED VIEW CONCURRENTLY monthly_stats;

-- Trigger (auto-update updated_at)
CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
  BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
  END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_orders_updated_at
  BEFORE UPDATE ON orders
  FOR EACH ROW EXECUTE FUNCTION set_updated_at();

Index Types & Creation
-- Default B-tree index (best for =, <, >, BETWEEN, LIKE 'prefix%')
CREATE INDEX        idx_users_email  ON users (email);
CREATE UNIQUE INDEX idx_users_email  ON users (email);

-- Composite index — column order matters!
-- Useful for: WHERE a=1 AND b=2, WHERE a=1, ORDER BY a,b
-- NOT useful for: WHERE b=2 alone (can't skip leading column)
CREATE INDEX idx_orders_user_status
  ON orders (user_id, status, created_at DESC);

-- Partial index — indexes only a subset of rows
CREATE INDEX idx_active_users
  ON users (email)
  WHERE active = true;     -- small, fast

-- Covering index — include non-key columns (index-only scan)
CREATE INDEX idx_orders_covering
  ON orders (user_id, created_at)
  INCLUDE (total, status);   -- PG 11+ / SQL Server

-- Expression index
CREATE INDEX idx_email_lower
  ON users (LOWER(email));    -- for LOWER(email) queries

-- Hash index (PG) — only = operator, smaller than B-tree
CREATE INDEX USING HASH ON sessions (token);

-- GIN index — full-text search, JSONB, arrays
CREATE INDEX idx_metadata_gin  ON posts USING GIN (metadata);
CREATE INDEX idx_body_fts      ON posts USING GIN (to_tsvector('english', body));

-- BRIN — Block Range INdex (huge append-only tables, time-series)
CREATE INDEX USING BRIN ON logs (created_at);

-- Concurrent build — no table lock
CREATE INDEX CONCURRENTLY idx_name ON big_table (col);

EXPLAIN ANALYZE — Reading Query Plans
✅Always run EXPLAIN ANALYZE (not just EXPLAIN) — it shows actual vs estimated row counts.
-- EXPLAIN: show plan without running
EXPLAIN
SELECT * FROM orders WHERE user_id = 42;

-- EXPLAIN ANALYZE: run AND show plan + timings
EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON)
SELECT * FROM orders WHERE user_id = 42;

-- Key terms in plan output:
-- Seq Scan     — full table scan (bad for large tables)
-- Index Scan   — uses index + fetches from heap
-- Index Only   — uses index alone (fastest — covering index)
-- Bitmap Scan  — combines multiple indexes
-- Nested Loop  — for small outer sets
-- Hash Join    — for larger unsorted sets
-- Merge Join   — pre-sorted inputs

-- cost=0.00..8.27 rows=1 width=248
--   ↑ start cost  ↑ total cost

-- actual time=0.123..0.124 rows=1 loops=1
-- Check: actual rows vs estimated rows
-- Large discrepancy = stale stats → ANALYZE table

-- Force index use (debug only)
SET enable_seqscan = off;
EXPLAIN SELECT * FROM orders WHERE user_id = 42;
RESET enable_seqscan;

Transactions — Patterns
-- Basic transaction
BEGIN;                         -- or START TRANSACTION
  UPDATE accounts
    SET balance = balance - 100
  WHERE id = 1;

  UPDATE accounts
    SET balance = balance + 100
  WHERE id = 2;
COMMIT;                        -- or ROLLBACK on error

-- Savepoints — partial rollback
BEGIN;
  INSERT INTO orders VALUES (...);
  SAVEPOINT after_order;

  INSERT INTO order_items VALUES (...);
  -- something goes wrong:
  ROLLBACK TO SAVEPOINT after_order;  -- undo items but keep order
  -- or: RELEASE SAVEPOINT after_order; (commit the savepoint)
COMMIT;

-- INSERT ... ON CONFLICT (upsert)
INSERT INTO users (email, name)
VALUES ('a@b.com', 'Alice')
ON CONFLICT (email)
DO UPDATE SET
  name       = EXCLUDED.name,
  updated_at = NOW()
RETURNING *;  -- get inserted/updated row back

-- DO NOTHING variant
ON CONFLICT (email) DO NOTHING;

Isolation Levels & Locking
READ UNCOMMITTEDDirty reads possible. Not recommended.
READ COMMITTEDDefault in PG/SQL Server. No dirty reads.
REPEATABLE READSame query returns same rows. Phantom reads possible.
SERIALIZABLEFull isolation. Slowest. No anomalies.
-- Set isolation level
BEGIN TRANSACTION ISOLATION LEVEL REPEATABLE READ;
BEGIN TRANSACTION ISOLATION LEVEL SERIALIZABLE;

-- Row-level locking
-- FOR UPDATE: lock rows for update, block others
SELECT * FROM inventory
WHERE product_id = 42
FOR UPDATE;                   -- blocks other SELECT FOR UPDATE

FOR UPDATE SKIP LOCKED;        -- skip locked rows (job queue pattern!)
FOR SHARE;                     -- allow other readers, block writers
FOR NO KEY UPDATE;             -- don't lock FK references

-- Advisory locks (application-level)
SELECT pg_advisory_lock(12345);     -- acquire
SELECT pg_advisory_unlock(12345);   -- release
SELECT pg_try_advisory_lock(12345)   -- non-blocking, returns bool

Job Queue Pattern — SKIP LOCKED
✅SKIP LOCKED is the standard pattern for building fair, concurrent job queues in PostgreSQL.
-- Job queue table
CREATE TABLE job_queue (
  id         BIGSERIAL PRIMARY KEY,
  payload    JSONB     NOT NULL,
  status     TEXT      DEFAULT 'pending',
  attempts   INT       DEFAULT 0,
  run_at     TIMESTAMPTZ DEFAULT NOW(),
  locked_by  TEXT
);

-- Worker: claim next available job atomically
WITH next_job AS (
  SELECT id
  FROM   job_queue
  WHERE  status  = 'pending'
    AND  run_at <= NOW()
  ORDER BY run_at
  LIMIT 1
  FOR UPDATE SKIP LOCKED   -- key: skip jobs taken by others
)
UPDATE job_queue
SET    status    = 'processing',
       locked_by = pg_backend_pid()::text,
       attempts  = attempts + 1
WHERE  id = (SELECT id FROM next_job)
RETURNING *;

JSONB — PostgreSQL
-- Access operators
metadata ->  'key'        -- returns JSON
metadata ->> 'key'        -- returns TEXT
metadata #>  '{a,b}'      -- nested path, JSON
metadata #>> '{a,b}'      -- nested path, TEXT

-- Containment operators (use GIN index!)
metadata @> '{"role":"admin"}'::jsonb  -- contains
metadata <@ '{"a":1,"b":2}'::jsonb     -- is contained by
metadata ?  'key'                        -- key exists
metadata ?| ARRAY['a','b']               -- any key exists
metadata ?& ARRAY['a','b']               -- all keys exist

-- Modify JSONB
jsonb_set(metadata, '{key}', '"value"')
jsonb_set(metadata, '{nested,key}', '42')
metadata || '{"new":"value"}'            -- merge
metadata - 'remove_key'                 -- delete key

-- Expand to rows
SELECT key, value
FROM  users, jsonb_each(metadata);

-- Extract to columns
SELECT jsonb_to_record(metadata)
  AS (name text, age int)
FROM users;

-- Aggregate to JSONB
jsonb_agg(row)

PIVOT / Crosstab
-- Pivot with CASE (works everywhere)
SELECT
  product,
  SUM(CASE WHEN month = 1 THEN revenue ELSE 0 END) AS jan,
  SUM(CASE WHEN month = 2 THEN revenue ELSE 0 END) AS feb,
  SUM(CASE WHEN month = 3 THEN revenue ELSE 0 END) AS mar
FROM  monthly_sales
GROUP BY product;

-- crosstab (PostgreSQL tablefunc extension)
SELECT * FROM crosstab(
  'SELECT product, month, revenue FROM monthly_sales ORDER BY 1,2'
) AS ct(product text, jan numeric, feb numeric);

-- SQL Server PIVOT
SELECT product, [1] AS jan, [2] AS feb, [3] AS mar
FROM monthly_sales
PIVOT (SUM(revenue) FOR month IN ([1],[2],[3])) pvt;

-- Full-text search (PostgreSQL)
SELECT title,
  ts_rank(to_tsvector('english',body), query) AS rank
FROM   posts,
       to_tsquery('english', 'laravel & database') query
WHERE  to_tsvector('english',body) @@ query
ORDER BY rank DESC;

⚠
Performance Anti-Patterns
🚫SELECT * — fetches unnecessary columns, prevents index-only scans, breaks with schema changes.
-- ❌ SELECT * — always name columns
SELECT * FROM users;
-- ✅
SELECT id, name, email FROM users;

-- ❌ Function on indexed column kills index
WHERE YEAR(created_at) = 2024
-- ✅ Use range instead
WHERE created_at >= '2024-01-01'
  AND created_at  < '2025-01-01'

-- ❌ Implicit type cast breaks index
WHERE user_id = '42'   -- user_id is INT
-- ✅
WHERE user_id = 42

-- ❌ OR can prevent index use
WHERE a = 1 OR b = 2
-- ✅ Use UNION ALL instead
SELECT * FROM t WHERE a = 1
UNION ALL
SELECT * FROM t WHERE b = 2

-- ❌ OFFSET is slow on large tables
LIMIT 20 OFFSET 100000  -- scans 100020 rows
-- ✅ Keyset pagination
WHERE id > last_seen_id ORDER BY id LIMIT 20

-- ❌ LIKE '%text%' — can't use B-tree index
-- ✅ Use full-text search or trigram index (pg_trgm)

VACUUM, ANALYZE & Maintenance
-- VACUUM: reclaim dead tuple space (runs auto by default)
VACUUM orders;              -- non-blocking
VACUUM FULL orders;         -- rewrites table, exclusive lock
VACUUM ANALYZE orders;      -- vacuum + update stats

-- ANALYZE: update query planner statistics
ANALYZE orders;             -- update stats for table
ANALYZE;                    -- update stats for all tables

-- Check table bloat
SELECT
  pg_size_pretty(pg_total_relation_size('orders')) AS total,
  pg_size_pretty(pg_relation_size('orders'))       AS table,
  pg_size_pretty(pg_indexes_size('orders'))        AS indexes;

-- Find unused indexes
SELECT indexrelname, idx_scan
FROM  pg_stat_user_indexes
WHERE idx_scan = 0
  AND schemaname = 'public';

-- Find slow queries (pg_stat_statements)
SELECT LEFT(query,80), calls,
  ROUND(mean_exec_time::numeric,2) AS mean_ms
FROM  pg_stat_statements
ORDER BY mean_exec_time DESC
LIMIT 10;

Table Partitioning
-- Range partitioning (PG 10+) — great for time-series
CREATE TABLE events (
  id         BIGSERIAL,
  created_at TIMESTAMPTZ NOT NULL,
  data       JSONB
) PARTITION BY RANGE (created_at);

-- Create partitions
CREATE TABLE events_2024_01
  PARTITION OF events
  FOR VALUES FROM ('2024-01-01') TO ('2024-02-01');

CREATE TABLE events_2024_02
  PARTITION OF events
  FOR VALUES FROM ('2024-02-01') TO ('2024-03-01');

-- List partitioning
PARTITION BY LIST (region);
FOR VALUES IN ('us-east', 'us-west');

-- Hash partitioning (even distribution)
PARTITION BY HASH (user_id);
FOR VALUES WITH (MODULUS 4, REMAINDER 0);

-- Detach old partition (no data loss, instant)
ALTER TABLE events DETACH PARTITION events_2024_01;
-- Drop old data instantly
DROP TABLE events_2024_01;

⚖
Cross-Dialect Cheatsheet
Auto-increment primary key
-- PostgreSQL
id BIGSERIAL PRIMARY KEY
-- or:
id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY

-- MySQL
id BIGINT AUTO_INCREMENT PRIMARY KEY

-- SQL Server
id BIGINT IDENTITY(1,1) PRIMARY KEY

-- SQLite
id INTEGER PRIMARY KEY AUTOINCREMENT
String concat
-- PG, SQLite, SQL Server:
first || ' ' || last
-- or SQL standard:
CONCAT(first, ' ', last)
-- MySQL: CONCAT only (|| is OR)
Top N rows
-- PG, MySQL 8, SQLite:
LIMIT 10
-- SQL Server:
SELECT TOP 10 * FROM t
-- Oracle:
FETCH FIRST 10 ROWS ONLY
Current timestamp
-- Standard SQL:
CURRENT_TIMESTAMP
-- PostgreSQL:
NOW()  -- returns TIMESTAMPTZ
-- MySQL:
NOW()  SYSDATE()
-- SQL Server:
GETDATE()  SYSDATETIME()
Upsert
-- PostgreSQL:
INSERT INTO t VALUES (...)
ON CONFLICT (key) DO UPDATE SET col = EXCLUDED.col;

-- MySQL:
INSERT INTO t VALUES (...)
ON DUPLICATE KEY UPDATE col = VALUES(col);

-- SQL Server:
MERGE target USING source ON (key)
WHEN MATCHED    THEN UPDATE SET ...
WHEN NOT MATCHED THEN INSERT ...;
String contains (case-insensitive)
-- PostgreSQL:
name ILIKE '%search%'
-- or:
LOWER(name) LIKE LOWER('%search%')
-- MySQL (case-insensitive by default):
name LIKE '%search%'
-- SQL Server:
name LIKE '%search%'  -- depends on collation
